```javascript
function deepFreeze(obj) {
  if (obj === null || obj === undefined || typeof obj !== 'object') return obj;
  if (Object.isFrozen(obj)) return obj;
  Object.freeze(obj);
  const keys = Object.keys(obj);
  for (let i = 0; i < keys.length; i++) {
    const val = obj[keys[i]];
    if (val !== null && typeof val === 'object' && !Object.isFrozen(val)) {
      deepFreeze(val);
    }
  }
  return obj;
}
```

    ```javascript
    // --- Lazy registry ----------------------------------------------------------------

    /**
     * Create a lazy-initialized registry. The initFn is not called until .get()
     * is first invoked. The result is cached for all subsequent .get() calls.
     *
     * @param {function} initFn - Zero-argument function that returns the registry value
     * @returns {{ get: function, initialized: boolean }} Registry accessor
     */
    function lazyRegistry(initFn) {
      const UNSET = Symbol('lazyRegistry.UNSET');
      let _value = UNSET;

      return {
        get() {
          if (_value === UNSET) {
            _value = initFn();
          }
          return _value;
        },
        get initialized() {
          return _value !== UNSET;
        },
      };
    }
    ```

    Key design decisions:
    - Uses a Symbol sentinel (`UNSET`) instead of null/undefined check, so initFn can legitimately return null or undefined and the cache still works. This is the critical design choice -- without it, returning null would cause re-initialization on every .get() call.
    - Returns a plain object with `.get()` method and `.initialized` getter -- simple, explicit API.
    - No error handling wrapping of initFn -- if it throws, the error propagates naturally and the next .get() call will retry (since _value is still UNSET). This is intentional: callers should handle initFn errors.
    - Zero dependencies, synchronous, pure CommonJS.
