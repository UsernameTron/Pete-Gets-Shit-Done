    ```javascript
    /**
     * Mapping of GSD agent to model for each profile.
     *
     * MODEL_PROFILES and VALID_PROFILES are lazily initialized on first access
     * via Object.defineProperty getters, then cached for all subsequent accesses.
     * This defers parsing cost until a caller actually needs agent metadata (PERF-03).
     */

    let _initCount = 0;
    let _modelProfiles = null;
    let _validProfiles = null;

    function _initialize() {
      if (_modelProfiles !== null) return;
      _initCount++;
      _modelProfiles = {
        'gsd-planner': { quality: 'opus', balanced: 'opus', budget: 'sonnet' },
        'gsd-roadmapper': { quality: 'opus', balanced: 'sonnet', budget: 'sonnet' },
        // ... (ALL current agents -- copy the full list from the existing file)
      };
      _validProfiles = Object.keys(_modelProfiles['gsd-planner']);
    }

    function formatAgentToModelMapAsTable(agentToModelMap) {
      // ... (unchanged -- same implementation)
    }

    function getAgentToModelMapForProfile(normalizedProfile) {
      _initialize();
      const agentToModelMap = {};
      for (const [agent, profileToModelMap] of Object.entries(_modelProfiles)) {
        agentToModelMap[agent] = profileToModelMap[normalizedProfile];
      }
      return agentToModelMap;
    }

    const _exports = {
      formatAgentToModelMapAsTable,
      getAgentToModelMapForProfile,
      _getInitCount: () => _initCount,
    };

    Object.defineProperty(_exports, 'MODEL_PROFILES', {
      get() {
        _initialize();
        return _modelProfiles;
      },
      enumerable: true,
      configurable: true,
    });

    Object.defineProperty(_exports, 'VALID_PROFILES', {
      get() {
        _initialize();
        return _validProfiles;
      },
      enumerable: true,
      configurable: true,
    });

    module.exports = _exports;
    ```

    Key design decisions:
    - Use `Object.defineProperty` with `get()` on the exports object so that `const { MODEL_PROFILES } = require(...)` triggers the getter at destructuring time (this is how JS destructuring works -- it reads the property).
    - `_initialize()` is guarded by `_modelProfiles !== null` so it only runs once.
    - `_getInitCount()` is a test-only helper (prefixed with underscore convention).
    - `getAgentToModelMapForProfile` calls `_initialize()` internally so it works even if MODEL_PROFILES was never directly accessed.
    - `formatAgentToModelMapAsTable` does NOT need `_initialize()` because it only operates on its argument (the agentToModelMap passed to it), not on the internal `_modelProfiles`.
    - `enumerable: true` ensures MODEL_PROFILES shows up in Object.keys(module.exports).
    - `configurable: true` allows test frameworks to mock if needed.

    IMPORTANT: Copy the COMPLETE, CURRENT list of agents from the existing model-profiles.cjs into the `_modelProfiles` assignment inside `_initialize()`. Do not truncate or abbreviate. The current file has the definitive agent list -- use all of them.
