    ```javascript
    // --- Token estimation -------------------------------------------------------------

    /**
     * Tokens-per-word ratio profiles for different model families.
     * Used by estimateTokens to approximate token counts without an external tokenizer.
     */
    const TOKEN_RATIOS = {
      default: 1.3,
      claude: 1.35,
      gpt: 1.3,
      code: 2.0,
    };

    /**
     * Approximate token count for a string using word-count heuristics.
     * Splits on whitespace, counts words, multiplies by a tokens-per-word ratio
     * selected by the optional model profile. Returns a rounded-up integer.
     *
     * @param {string} text - Input string to estimate tokens for
     * @param {object} [opts] - Options
     * @param {string} [opts.model] - Model profile name (key in TOKEN_RATIOS). Defaults to 'default'.
     * @returns {number} Approximate token count (integer, 0 for empty/null/undefined input)
     */
    function estimateTokens(text, opts) {
      if (!text || typeof text !== 'string') return 0;
      const words = text.trim().split(/\s+/);
      if (words.length === 1 && words[0] === '') return 0;
      const model = (opts && opts.model) || 'default';
      const ratio = TOKEN_RATIOS[model] || TOKEN_RATIOS.default;
      return Math.ceil(words.length * ratio);
    }
    ```

    Key design decisions:
    - Uses `Math.ceil` for rounding so estimates are never lower than actual (conservative). This is standard practice for budget calculations -- you want to slightly overcount rather than undercount.
    - `trim().split(/\s+/)` handles multiple spaces, tabs, newlines between words.
    - The guard `words.length === 1 && words[0] === ''` catches whitespace-only input after trim produces `['']`.
    - `TOKEN_RATIOS` is a plain object constant, not exported -- it is an implementation detail. Only `estimateTokens` is part of the public API.
    - Falls back to `TOKEN_RATIOS.default` for unknown model profiles rather than throwing -- fail-safe behavior.
    - No character-based fallback needed: the whitespace split handles short strings fine (a single character word still counts as 1 word).
